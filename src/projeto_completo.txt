package ong.eventos;


import ong.usuario.Voluntario;

import java.util.ArrayList;
import java.util.List;


public abstract class Atividade{
    private int id;
    private String titulo;
    private String descricao;
    private String data;
    private String local;
    private int vagasDisponiveis;
    private List<Voluntario> voluntariosInscritos;
    
    public Atividade(int id, String titulo, String descricao, String data, String local, int vagasDisponiveis){
        this.id = id;
        this.titulo = titulo;
        this.descricao = descricao;
        this.data = data;
        this.local = local;
        this.vagasDisponiveis = vagasDisponiveis;
        this.voluntariosInscritos = new ArrayList<>();

    }
    
    //vou testar fazer a inscricao aqui, mas se 
    // fica mto baguncado eu mudo depois.
    public boolean inscreverVoluntario(Voluntario voluntario){
        if(vagasDisponiveis > 0 && !voluntariosInscritos.contains(voluntario)){
            voluntariosInscritos.add(voluntario);
            vagasDisponiveis--;
            voluntario.addAtividadeAtiva(this);
            return true;
        }
        return false;
    }

    public int getId(){
        return this.id;
    }
    
    public String getTitulo(){
        return this.titulo;
    }

    public String getData(){
        return this.data;
    }

    public String getLocal(){
        return this.local;
    }

    public int getVagasDisponiveis(){
        return this.vagasDisponiveis;
    }

}

                abrirDialogoAdocao(janela, selecionado);
            }
        });

        painelBotoes.add(btnVerDetalhes);
        painelBotoes.add(btnSolicitarAdocao);

        painelPrincipal.add(painelTopo, BorderLayout.NORTH);
        painelPrincipal.add(scrollTabela, BorderLayout.CENTER);
        painelPrincipal.add(painelBotoes, BorderLayout.SOUTH);

        atualizarTabela();

        janela.setConteudo(painelPrincipal);
        janela.setVisible(true);
    }

    private void atualizarTabela() {
        modeloTabela.setRowCount(0);
        String termo = txtPesquisar.getText().trim().toLowerCase();

        List<Animal> listaFiltrada;
        if (filtroEspecieAtual.equals("Cachorro") || filtroEspecieAtual.equals("Gato")) {
            listaFiltrada = inv.filtrarPorEspecie(filtroEspecieAtual);
        } else if (filtroEspecieAtual.equals("Outros")) {
            listaFiltrada = new ArrayList<>();
            for (Animal a : inv.getAnimais()) {
                if (!(a instanceof Cachorro) && !(a instanceof Gato)) {
                    listaFiltrada.add(a);
                }
            }
        } else {
            listaFiltrada = inv.getAnimais();
        }

        for (Animal a : listaFiltrada) {
            String status = a.getStatusAdocao().toLowerCase();
            if (!status.contains("disponivel") && !status.contains("disponível")) {
                continue;
            }
            if (termo.isEmpty() || a.getNome().toLowerCase().contains(termo)) {
                String idFormatado = String.format("%02d", a.getId());
                String idadeFormatada = a.getIdade() + (a.getIdade() == 1 ? " ano" : " anos");

                modeloTabela.addRow(new Object[]{
                    idFormatado,
                    a.getNome(),
                    a.getEspecie(),
                    idadeFormatada,
                    a.getStatusAdocao()
                });
            }
        }
    }

    private Animal getAnimalSelecionado(Component parent) {
        int linha = tabAnimais.getSelectedRow();
        if (linha == -1) {
            JOptionPane.showMessageDialog(parent, "Selecione um animal na tabela primeiro.", "Aviso", JOptionPane.WARNING_MESSAGE);
            return null;
        }
        int idAnimal = Integer.parseInt(modeloTabela.getValueAt(linha, 0).toString());
        return inv.buscarPorId(idAnimal);
    }

    private void abrirDialogoAdocao(Component parent, Animal animal) {
        CustomTextField txtNomeCandidato = new CustomTextField();
        CustomTextField txtCpfCandidato = new CustomTextField();

        if (voluntarioLogado != null) {
            txtNomeCandidato.setText(voluntarioLogado.getNome());
            txtNomeCandidato.setEditable(false);
        }

        Object[] campos = {
            new CustomLabel("Animal selecionado: " + animal.getNome() + " (ID: " + String.format("%02d", animal.getId()) + ")", Font.BOLD, 16),
            new CustomLabel("Nome do Candidato:"), txtNomeCandidato,
            new CustomLabel("CPF do Candidato:"), txtCpfCandidato
        };

        int opcao = JOptionPane.showConfirmDialog(parent, campos, "Solicitar Adoção", JOptionPane.OK_CANCEL_OPTION);
        if (opcao == JOptionPane.OK_OPTION) {
            String nome = txtNomeCandidato.getText().trim();
            String cpf = txtCpfCandidato.getText().trim();

            if (nome.isEmpty() || cpf.isEmpty()) {
                JOptionPane.showMessageDialog(parent, "Preencha nome e CPF.", "Erro", JOptionPane.ERROR_MESSAGE);
                return;
            }

            try {
                int idSolicitacao = (int) (System.currentTimeMillis() % 10000);
                boolean sucesso = inv.solicitarAdocao(idSolicitacao, animal.getId(), nome, cpf);

                if (sucesso) {
                    JOptionPane.showMessageDialog(
                        parent,
                        "Solicitação #" + String.format("%04d", idSolicitacao) + " registrada com sucesso para " + animal.getNome() + "!",
                        "Sucesso",
                        JOptionPane.INFORMATION_MESSAGE
                    );
                    atualizarTabela();
                }
            } catch (IllegalArgumentException | IllegalStateException ex) {
                JOptionPane.showMessageDialog(parent, ex.getMessage(), "Não foi possível solicitar", JOptionPane.WARNING_MESSAGE);
            }
        }
    }

    public boolean solicitarAdocao(int idSolicitacao, int idAnimal, String nomeCandidato, String cpfCandidato){
        Animal animal = buscarPorId(idAnimal);
        if(animal == null){
            throw new IllegalArgumentException("Erro: Animal nao encontrado.");
        }

        if(animal.getEstadoSaude().equalsIgnoreCase("Critico") || animal.getEstadoSaude().equalsIgnoreCase("Tratamento Critico")){
            throw new IllegalStateException("Erro: O animal " + animal.getNome() + " esta em tratamento critico e nao pode ser adotado.");
        }

        if (!animal.getStatusAdocao().equalsIgnoreCase("Disponivel") && !animal.getStatusAdocao().equalsIgnoreCase("Disponível")) {
            throw new IllegalStateException("Erro: O animal " + animal.getNome() + " esta indisponivel para adocao (Status: " + animal.getStatusAdocao() + ").");
        }

        SolicitacaoAdocao novaSol = new SolicitacaoAdocao(idSolicitacao, animal, nomeCandidato, cpfCandidato);
        solicitacoes.add(novaSol);
        System.out.println("Solicitacao de adocao criada para o(a) " + animal.getNome() + " em nome de " + nomeCandidato);
        return true;

package ong.eventos;

import ong.usuario.Voluntario;
import java.util.ArrayList;
import java.util.List;

public abstract class Atividade {
    private int id;
    private String titulo;
    private String descricao;
    private String data;
    private String local;
    private int vagasDisponiveis;
    private List<Voluntario> voluntariosInscritos;

    public Atividade(int id, String titulo, String descricao, String data, String local, int vagasDisponiveis) {
        this.id = id;
        this.titulo = titulo;
        this.descricao = descricao;
        this.data = data;
        this.local = local;
        this.vagasDisponiveis = vagasDisponiveis;
        this.voluntariosInscritos = new ArrayList<>();
    }

    public boolean inscreverVoluntario(Voluntario voluntario) {
        if (vagasDisponiveis > 0 && !voluntariosInscritos.contains(voluntario)) {
            voluntariosInscritos.add(voluntario);
            vagasDisponiveis--;
            voluntario.addAtividadeAtiva(this);
            return true;
        }
        return false;
    }

    public int getId() {
        return this.id;
    }

    public String getTitulo() {
        return this.titulo;
    }

    public String getDescricao() {
        return this.descricao;
    }

    public String getData() {
        return this.data;
    }

    public String getLocal() {
        return this.local;
    }

    public int getVagasDisponiveis() {
        return this.vagasDisponiveis;
    }

    public List<Voluntario> getVoluntariosInscritos() {
        return this.voluntariosInscritos;
    }

    public String getTipo() {
        return this.getClass().getSimpleName();
    }
}

    private void abrirJanelaGerenciarAnimais() {
        CustomFrame tela = new CustomFrame("Gestão de Animais - Abrigo Amigos Fofos", 820, 580);

        JPanel painelPrincipal = new JPanel(new BorderLayout(16, 16));
        painelPrincipal.setPreferredSize(new Dimension(780, 520));
        painelPrincipal.setBorder(BorderFactory.createEmptyBorder(16, 24, 16, 24));
        painelPrincipal.setOpaque(false);

        CustomLabel lblTitulo = new CustomLabel("GESTÃO DE ANIMAIS", Font.BOLD, 22);

        String[] colunas = {"ID", "Nome", "Espécie", "Raça", "Idade", "Porte", "Saúde", "Status"};
        DefaultTableModel modelo = new DefaultTableModel(colunas, 0) {
            @Override
            public boolean isCellEditable(int row, int column) {
                return false;
            }
        };

        Runnable atualizar = () -> {
            modelo.setRowCount(0);
            for (Animal a : inv.getAnimais()) {
                modelo.addRow(new Object[]{
                    String.format("%02d", a.getId()),
                    a.getNome(),
                    a.getEspecie(),
                    a.getRaca(),
                    a.getIdade(),
                    a.getPorte(),
                    a.getEstadoSaude(),
                    a.getStatusAdocao()
                });
            }
        };
        atualizar.run();

        JTable tabela = new JTable(modelo);
        tabela.setSelectionMode(ListSelectionModel.SINGLE_SELECTION);
        tabela.setRowHeight(26);
        tabela.getTableHeader().setFont(new Font("Quicksand", Font.BOLD, 12));
        tabela.setFont(new Font("Quicksand", Font.PLAIN, 12));

        JPanel painelBotoes = new JPanel(new FlowLayout(FlowLayout.CENTER, 12, 4));
        painelBotoes.setOpaque(false);

        CustomBotao btnCadastrar = new CustomBotao("Cadastrar Animal", e -> {
            abrirFormularioAnimal(tela, null, atualizar);
        });
        btnCadastrar.setFont(new Font("Quicksand", Font.BOLD, 15));

        CustomBotao btnEditar = new CustomBotao("Editar Selecionado", e -> {
            int linha = tabela.getSelectedRow();
            if (linha == -1) {
                JOptionPane.showMessageDialog(tela, "Selecione um animal na tabela.", "Aviso", JOptionPane.WARNING_MESSAGE);
                return;
            }
            int id = Integer.parseInt(modelo.getValueAt(linha, 0).toString());
            Animal alvo = inv.buscarPorId(id);
            if (alvo != null) {
                abrirFormularioAnimal(tela, alvo, atualizar);
            }
        });
        btnEditar.setFont(new Font("Quicksand", Font.BOLD, 15));

        CustomBotao btnRemover = new CustomBotao("Remover Selecionado", e -> {
            int linha = tabela.getSelectedRow();
            if (linha == -1) {
                JOptionPane.showMessageDialog(tela, "Selecione um animal na tabela.", "Aviso", JOptionPane.WARNING_MESSAGE);
                return;
            }
            int id = Integer.parseInt(modelo.getValueAt(linha, 0).toString());
            boolean removido = inv.removerAnimal(id);
            if (removido) {
                JOptionPane.showMessageDialog(tela, "Animal removido do sistema.");
                atualizar.run();
            } else {
                JOptionPane.showMessageDialog(tela, "Não é possível remover um animal já adotado.", "Operação Bloqueada", JOptionPane.ERROR_MESSAGE);
            }
        });
        btnRemover.setFont(new Font("Quicksand", Font.BOLD, 15));

        CustomBotao btnFechar = new CustomBotao("Fechar", e -> tela.dispose());
        btnFechar.setFont(new Font("Quicksand", Font.BOLD, 15));

        painelBotoes.add(btnCadastrar);
        painelBotoes.add(btnEditar);
        painelBotoes.add(btnRemover);
        painelBotoes.add(btnFechar);

        painelPrincipal.add(lblTitulo, BorderLayout.NORTH);
        painelPrincipal.add(new JScrollPane(tabela), BorderLayout.CENTER);
        painelPrincipal.add(painelBotoes, BorderLayout.SOUTH);

        tela.setConteudo(painelPrincipal);
        tela.setVisible(true);
    }

    private void abrirFormularioAnimal(Component parent, Animal existente, Runnable aoSalvar) {
        String[] especies = {"Cachorro", "Gato", "Outros"};
        JComboBox<String> cbEspecie = new JComboBox<>(especies);
        CustomTextField txtNome = new CustomTextField();
        CustomTextField txtRaca = new CustomTextField();
        CustomTextField txtIdade = new CustomTextField();
        CustomTextField txtPorte = new CustomTextField();
        CustomTextField txtSaude = new CustomTextField();
        CustomTextField txtData = new CustomTextField();
        CustomTextField txtStatus = new CustomTextField();

        if (existente != null) {
            cbEspecie.setSelectedItem(existente.getEspecie());
            cbEspecie.setEnabled(false);
            txtNome.setText(existente.getNome());
            txtRaca.setText(existente.getRaca());
            txtIdade.setText(String.valueOf(existente.getIdade()));
            txtPorte.setText(existente.getPorte());
            txtSaude.setText(existente.getEstadoSaude());
            txtData.setText(existente.getDataResgate());
            txtStatus.setText(existente.getStatusAdocao());
        } else {
            txtStatus.setText("Disponível");
        }

        Object[] campos = {
            new CustomLabel("Espécie:"), cbEspecie,
            new CustomLabel("Nome:"), txtNome,
            new CustomLabel("Raça:"), txtRaca,
            new CustomLabel("Idade (anos):"), txtIdade,
            new CustomLabel("Porte:"), txtPorte,
            new CustomLabel("Estado de Saúde:"), txtSaude,
            new CustomLabel("Data de Resgate (dd/mm/aaaa):"), txtData,
            new CustomLabel("Status de Adoção:"), txtStatus
        };

        String titulo = existente == null ? "Cadastrar Animal Resgatado" : "Editar Registro de Animal";
        if (JOptionPane.showConfirmDialog(parent, campos, titulo, JOptionPane.OK_CANCEL_OPTION) == JOptionPane.OK_OPTION) {
            try {
                String nome = txtNome.getText().trim();
                String raca = txtRaca.getText().trim();
                int idade = Integer.parseInt(txtIdade.getText().trim());
                String porte = txtPorte.getText().trim();
                String saude = txtSaude.getText().trim();
                String data = txtData.getText().trim();
                String status = txtStatus.getText().trim();

                if (nome.isEmpty() || raca.isEmpty() || porte.isEmpty() || saude.isEmpty() || data.isEmpty() || status.isEmpty()) {
                    JOptionPane.showMessageDialog(parent, "Preencha todos os campos.", "Aviso", JOptionPane.WARNING_MESSAGE);
                    return;
                }

                if (existente == null) {
                    int novoId = 1;
                    for (Animal a : inv.getAnimais()) {
                        if (a.getId() >= novoId) {
                            novoId = a.getId() + 1;
                        }
                    }
                    String esp = (String) cbEspecie.getSelectedItem();
                    Animal novo;
                    if ("Cachorro".equals(esp)) {
                        novo = new Cachorro(novoId, nome, raca, idade, porte, saude, data, status);
                    } else if ("Gato".equals(esp)) {
                        novo = new Gato(novoId, nome, raca, idade, porte, saude, data, status);
                    } else {
                        novo = new Outros(novoId, nome, raca, idade, porte, saude, data, status);
                    }
                    inv.addAnimal(novo);
                } else {
                    inv.editarAnimal(existente.getId(), nome, raca, idade, porte, saude, data, status);
                }
                aoSalvar.run();
            } catch (NumberFormatException ex) {
                JOptionPane.showMessageDialog(parent, "A idade deve ser um número inteiro.", "Erro", JOptionPane.ERROR_MESSAGE);
            }
        }

    private void atualizarTabela() {
        modeloTabela.setRowCount(0);
        String termo = txtPesquisar.getText().trim().toLowerCase();

        List<Animal> listaFiltrada;
        if (filtroEspecieAtual.equals("Cachorro") || filtroEspecieAtual.equals("Gato")) {
            listaFiltrada = inv.filtrarPorEspecie(filtroEspecieAtual);
        } else if (filtroEspecieAtual.equals("Outros")) {
            listaFiltrada = new ArrayList<>();
            for (Animal a : inv.getAnimais()) {
                if (!(a instanceof Cachorro) && !(a instanceof Gato)) {
                    listaFiltrada.add(a);
                }
            }
        } else {
            listaFiltrada = inv.getAnimais();
        }

        for (Animal a : listaFiltrada) {
            String status = a.getStatusAdocao().toLowerCase();
            if (!status.contains("disponivel") && !status.contains("disponível")) {
                continue;
            }
            if (termo.isEmpty() || a.getNome().toLowerCase().contains(termo)) {
                String idFormatado = String.format("%02d", a.getId());
                String idadeFormatada = a.getIdade() + (a.getIdade() == 1 ? " ano" : " anos");

                modeloTabela.addRow(new Object[]{
                    idFormatado,
                    a.getNome(),
                    a.getEspecie(),
                    idadeFormatada,
                    a.getStatusAdocao()
                });
            }
        }
    }

    private Animal getAnimalSelecionado(Component parent) {

        if (linha == -1) {
            JOptionPane.showMessageDialog(parent, "Selecione um animal na tabela primeiro.", "Aviso", JOptionPane.WARNING_MESSAGE);
            return null;
        }
        int idAnimal = Integer.parseInt(modeloTabela.getValueAt(linha, 0).toString());
        return inv.buscarPorId(idAnimal);
    }

    private void abrirDialogoAdocao(Component parent, Animal animal) {
        CustomTextField txtNomeCandidato = new CustomTextField();
        CustomTextField txtCpfCandidato = new CustomTextField();

        if (voluntarioLogado != null) {
            txtNomeCandidato.setText(voluntarioLogado.getNome());
            txtNomeCandidato.setEditable(false);
        }

        Object[] campos = {
            new CustomLabel("Animal selecionado: " + animal.getNome() + " (ID: " + String.format("%02d", animal.getId()) + ")", Font.BOLD, 16),
            new CustomLabel("Nome do Candidato:"), txtNomeCandidato,
            new CustomLabel("CPF do Candidato:"), txtCpfCandidato
        };

        int opcao = JOptionPane.showConfirmDialog(parent, campos, "Solicitar Adoção", JOptionPane.OK_CANCEL_OPTION);
        if (opcao == JOptionPane.OK_OPTION) {
            String nome = txtNomeCandidato.getText().trim();
            String cpf = txtCpfCandidato.getText().trim();

            if (nome.isEmpty() || cpf.isEmpty()) {
                JOptionPane.showMessageDialog(parent, "Preencha nome e CPF.", "Erro", JOptionPane.ERROR_MESSAGE);
                return;
            }

            if (voluntarioLogado != null) {
                String cpfDigitadoLimpo = cpf.replaceAll("[^0-9]", "");
                String cpfVoluntarioLimpo = voluntarioLogado.getCpf().replaceAll("[^0-9]", "");
                if (!cpfDigitadoLimpo.equals(cpfVoluntarioLimpo)) {
                    JOptionPane.showMessageDialog(parent, "O CPF informado não corresponde ao do voluntário logado.", "CPF Inválido", JOptionPane.ERROR_MESSAGE);
                    return;
                }
                cpf = voluntarioLogado.getCpf();
            }

            try {
                int idSolicitacao = (int) (System.currentTimeMillis() % 10000);
                boolean sucesso = inv.solicitarAdocao(idSolicitacao, animal.getId(), nome, cpf);

                if (sucesso) {
                    JOptionPane.showMessageDialog(
                        parent,
                        "Solicitação #" + String.format("%04d", idSolicitacao) + " registrada com sucesso para " + animal.getNome() + "!",
                        "Sucesso",
                        JOptionPane.INFORMATION_MESSAGE
                    );
                    atualizarTabela();
                }
            } catch (IllegalArgumentException | IllegalStateException ex) {
                JOptionPane.showMessageDialog(parent, ex.getMessage(), "Não foi possível solicitar", JOptionPane.WARNING_MESSAGE);
            }
        }

    private void exibirProntuarioCompleto(Component parent, int idAnimal) {
        Animal animal = inv.buscarPorId(idAnimal);
        if (animal == null) {
            JOptionPane.showMessageDialog(parent, "Animal não encontrado.", "Erro", JOptionPane.ERROR_MESSAGE);
            return;
        }

        StringBuilder sb = new StringBuilder();
        sb.append("PRONTUÁRIO MÉDICO - ").append(animal.getNome().toUpperCase()).append("\n");
        sb.append("ID: ").append(String.format("%02d", animal.getId()))
          .append(" | Espécie: ").append(animal.getEspecie())
          .append(" | Raça: ").append(animal.getRaca()).append("\n");
        sb.append("Estado de Saúde Atual: ").append(animal.getEstadoSaude()).append("\n\n");

        sb.append("=== CONSULTAS E MEDICAMENTOS ===\n");
        if (animal.getHistoricoConsultas().isEmpty()) {
            sb.append("Nenhuma consulta registrada.\n");
        } else {
            for (Consulta c : animal.getHistoricoConsultas()) {
                sb.append("#").append(c.getIdConsulta()).append(" [").append(c.getData()).append("] Dr(a). ")
                  .append(c.getVeterinario().getNome()).append("\nDiagnóstico: ").append(c.getDiagnostico())
                  .append(" | Prescrição: ").append(c.getPrescricao()).append("\n\n");
            }
        }

        sb.append("=== CIRURGIAS E TRATAMENTOS ===\n");
        if (animal.getHistoricoProcedimentos().isEmpty()) {
            sb.append("Nenhum procedimento registrado.\n");
        } else {
            for (ProcedimentoMedico p : animal.getHistoricoProcedimentos()) {
                sb.append("#").append(p.getIdProcedimento()).append(" [").append(p.getData()).append("] ")
                  .append(p.getTipo()).append(" - ").append(p.getStatus()).append("\nDescrição: ")
                  .append(p.getDescricao()).append(" (Dr(a). ").append(p.getVeterinario().getNome()).append(")\n\n");
            }
        }

        sb.append("=== CARTEIRA DE VACINAÇÃO ===\n");
        if (animal.getCarteiraVacinacao().isEmpty()) {
            sb.append("Nenhuma vacina registrada.\n");
        } else {
            for (Vacina v : animal.getCarteiraVacinacao()) {
                sb.append("#").append(v.getIdVacina()).append(" ").append(v.getNomeVacina())
                  .append(" (").append(v.getDose()).append(") - Aplicada: ").append(v.getDataAplicacao())
                  .append(" | Próxima: ").append(v.getDataProximaDose()).append("\n");
            }
        }

        JTextArea area = new JTextArea(sb.toString(), 22, 52);
        area.setFont(new Font("Quicksand", Font.PLAIN, 13));
        area.setEditable(false);
        area.setLineWrap(true);
        area.setWrapStyleWord(true);

        JOptionPane.showMessageDialog(parent, new JScrollPane(area), "Relatório de Saúde", JOptionPane.INFORMATION_MESSAGE);
    }

    public ArrayList<Notificacao> getNotificacoesParaUsuario(Usuario user) {
        ArrayList<Notificacao> resultado = new ArrayList<>();
        DateTimeFormatter fmt = DateTimeFormatter.ofPattern("dd/MM/yyyy");
        LocalDate hoje = LocalDate.now();

        for (Animal a : animais) {
            for (Vacina v : a.getCarteiraVacinacao()) {
                try {
                    LocalDate dataProx = LocalDate.parse(v.getDataProximaDose(), fmt);
                    long diasRestantes = ChronoUnit.DAYS.between(hoje, dataProx);
                    if (diasRestantes <= 30) {
                        String textoDias = diasRestantes < 0
                            ? "vencida há " + Math.abs(diasRestantes) + " dia(s)"
                            : "vence em " + diasRestantes + " dia(s) (" + v.getDataProximaDose() + ")";
                        resultado.add(new Notificacao(
                            0,
                            "Vacina Próxima do Vencimento",
                            "Animal " + a.getNome() + " (ID " + String.format("%02d", a.getId()) + ") - Vacina " + v.getNomeVacina() + " " + textoDias + ".",
                            dataHoje(),
                            null
                        ));
                    }
                } catch (Exception ignored) {
                }
            }
        }

        for (int i = notificacoes.size() - 1; i >= 0; i--) {
            Notificacao n = notificacoes.get(i);
            if (n.getCpfDestinatario() == null ||
                n.getCpfDestinatario().equals(user.getCpf()) ||
                user instanceof Administrador) {
                resultado.add(n);
            }

        }

        SolicitacaoAdocao novaSol = new SolicitacaoAdocao(idSolicitacao, animal, nomeCandidato, cpfCandidato);

            );

            for (SolicitacaoAdocao outra : solicitacoes) {
                if (outra.getIdSolicitacao() != idSolicitacao &&
                    outra.getAnimal().getId() == sol.getAnimal().getId() &&

        int gatos = 0;
        int outros = 0;
        for (Animal a : animais) {
            if (a instanceof Cachorro) {
                caes++;
            } else if (a instanceof Gato) {
                gatos++;
            } else {
                outros++;
            }
        }
        if (caes >= gatos && caes >= outros) {
            return "Cachorro (Cachorros: " + caes + " | Gatos: " + gatos + " | Outros: " + outros + ")";

            return "Gato (Cachorros: " + caes + " | Gatos: " + gatos + " | Outros: " + outros + ")";

            return "Outros (Cachorros: " + caes + " | Gatos: " + gatos + " | Outros: " + outros + ")";
        }
    }

package ong.sistema;

import java.util.ArrayList;
import ong.usuario.*;

public class SistemaUsuario {
    private ArrayList<Usuario> todosUsuarios = new ArrayList<>();

    public void cadastrarUsuario(Usuario u) {
        todosUsuarios.add(u);
    }

    public ArrayList<Usuario> getTodosUsuarios() {
        return todosUsuarios;
    }

    public boolean removerUsuario(String email) {
        Usuario alvo = buscarPorEmail(email);
        if (alvo != null) {
            todosUsuarios.remove(alvo);
            return true;
        }
        return false;
    }

    public boolean promoverParaAdmin(String email) {
        Usuario alvo = buscarPorEmail(email);
        if (alvo != null && !(alvo instanceof Administrador)) {
            Administrador novoAdmin = new Administrador(alvo.getNome(), alvo.getEmail(), alvo.getSenha(), alvo.getCpf());
            todosUsuarios.remove(alvo);
            todosUsuarios.add(novoAdmin);
            return true;
        }
        return false;
    }

    public Usuario buscarPorEmail(String email) {
        for (Usuario u : todosUsuarios) {
            if (validarEmail(u, email)) {
                return u;
            }
        }
        return null;
    }

    public void mostrarUsuarios() {
        for (Usuario u : todosUsuarios) {
            System.out.printf(u.getNome() + " - email: " + u.getEmail() + ".\n");
        }
    }

    public Usuario validarLogin(String email, String senha) {
        for (Usuario u : todosUsuarios) {
            if (validarEmail(u, email) && u.validarSenha(senha)) {
                return u;
            }
        }
        return null;
    }

    public boolean validarEmail(Usuario u, String email) {
        return u.getEmail().equalsIgnoreCase(email.trim());
    }

    public int contarVoluntariosAtivos() {
        int total = 0;
        for (Usuario u : todosUsuarios) {
            if (u instanceof Voluntario) {
                total++;
            }
        }
        return total;
    }
}

import java.awt.*;

public class CustomFrame extends JFrame{
    private JPanel painelPrincipal;

    public CustomFrame(String titulo, int largura, int altura){
        setTitle(titulo);
        setSize(largura, altura);
        setDefaultCloseOperation(JFrame.DISPOSE_ON_CLOSE);
        setLocationRelativeTo(null);

        painelPrincipal = new JPanel(new GridBagLayout());
        painelPrincipal.setBackground(Color.WHITE);

        add(painelPrincipal);

    }

    public void setConteudo(JComponent componente){
        painelPrincipal.removeAll();
        painelPrincipal.add(componente, new GridBagConstraints());
        revalidate();
        repaint();
    }
}

package ong.usuario;

public abstract class Usuario {
    private String nome;
    private String email;
    private String senha;
    private final String CPF;

    public Usuario(String nome, String email, String senha, String CPF) {
        this.nome = nome;
        this.email = email;
        this.senha = senha;
        this.CPF = CPF;
    }

    public String getNome() {
        return this.nome;
    }

    public String getEmail() {
        return this.email;
    }

    public String getSenha() {
        return this.senha;
    }

    public String getCpf() {
        return this.CPF;
    }

    public boolean validarSenha(String senhaDigitada) {
        return this.senha != null && this.senha.equals(senhaDigitada);
    }
}

package ong.usuario;

import ong.animais.Animal;
import ong.eventos.Atividade;
import java.util.ArrayList;
import java.util.List;

public class Voluntario extends Usuario {
    private List<Atividade> historico = new ArrayList<>();
    private List<Atividade> atvAtivas = new ArrayList<>();
    private List<Animal> animaisResponsavel = new ArrayList<>();

    public Voluntario(String nome, String email, String senha, String CPF) {
        super(nome, email, senha, CPF);
    }

    public List<Atividade> getHistorico() {
        return this.historico;
    }

    public List<Atividade> getAtvAtivas() {
        return this.atvAtivas;
    }

    public void addAtividadeAtiva(Atividade atv) {
        this.atvAtivas.add(atv);
    }

    public void concluirAtividadeAtiva(Atividade atv) {
        if (this.atvAtivas.contains(atv)) {
            this.historico.add(atv);
            this.atvAtivas.remove(atv);
        }
    }

    public void addAnimalResponsavel(Animal animal) {
        if (!this.animaisResponsavel.contains(animal)) {
            this.animaisResponsavel.add(animal);
        }
    }

    public List<Animal> getAnimaisResponsavel() {
        return this.animaisResponsavel;
    }
}
